function doPost(e) { var sheets = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); var date = data.date; var action = data.action; var time = data.time || ""; var operator = data.operator || ""; var note = data.note || ""; if (action === "gameScore") { var scoreSheet = sheets.getSheetByName("遊戲積分紀錄"); if (!scoreSheet) { scoreSheet = sheets.insertSheet("遊戲積分紀錄"); scoreSheet.appendRow(["日期", "時間", "玩家", "獲得分數", "得分原因"]); } var score = data.score || 0; scoreSheet.appendRow([date, time, operator, score, note]); return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON); } if (action === "sendChat") { var chatSheet = sheets.getSheetByName("Chat"); if (!chatSheet) { chatSheet = sheets.insertSheet("Chat"); chatSheet.appendRow(["日期", "時間", "角色", "訊息"]); } chatSheet.appendRow([date, time, operator, data.message]); return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON); } var sheet = sheets.getSheets()[0]; if (action === "checkin") { sheet.appendRow([date, "已回家", note, time, operator]); } else if (action === "cancel") { var rows = sheet.getDataRange().getValues(); for (var i = rows.length - 1; i >= 1; i--) { var cellDateStr = _formatDateSafe(rows[i][0]); if (cellDateStr === date) { sheet.deleteRow(i + 1); } } } return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON); } function doGet(e) { try { var sheets = SpreadsheetApp.getActiveSpreadsheet(); var targetDate = e.parameter.date; var callback = e.parameter.callback; var reqType = e.parameter.type || "checkin"; // ============================ // 1. 讀取所有聊天紀錄 // ============================ if (reqType === "chat") { var chatSheet = sheets.getSheetByName("Chat"); var messages = []; if (chatSheet) { var rows = chatSheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { var cellDateStr = _formatDateSafe(rows[i][0]); if (targetDate === "all" || cellDateStr === targetDate) { messages.push({ date: cellDateStr, time: _formatTimeSafe(rows[i][1]), user: String(rows[i][2] || "").trim(), message: String(rows[i][3] || "").trim() }); } } } return _jsonpResponse(callback, messages); } // ============================ // 🔔 2. 快速算總留言數 (給首頁紅點用) // ============================ if (reqType === "chatCount") { var chatSheet = sheets.getSheetByName("Chat"); var count = 0; if (chatSheet) { // 扣除第一行的標題，剩下的就是訊息總數 count = Math.max(0, chatSheet.getLastRow() - 1); } return _jsonpResponse(callback, { count: count }); } // ============================ // 3. 讀取打卡紀錄 // ============================ var sheet = sheets.getSheets()[0]; var rows = sheet.getDataRange().getValues(); var result = { checked: false, debug: "No data found" }; for (var i = rows.length - 1; i >= 1; i--) { var cellDateStr = _formatDateSafe(rows[i][0]); if (cellDateStr === targetDate) { var status = String(rows[i][1] || "").trim(); if (status.indexOf("已回家") !== -1) { result = { checked: true, note: String(rows[i][2] || "").trim(), time: _formatTimeSafe(rows[i][3]), operator: String(rows[i][4] || "").trim() }; } break; } } return _jsonpResponse(callback, result); } catch (error) { return _jsonpResponse(callback, { error: error.toString() }); } } function _formatDateSafe(val) { if (!val) return ""; var d = new Date(val); if (!isNaN(d.getTime())) return Utilities.formatDate(d, "GMT+8", "yyyy-MM-dd"); return String(val).trim().replace(/\//g, "-"); } function _formatTimeSafe(val) { if (!val) return ""; if (val instanceof Date) return Utilities.formatDate(val, "GMT+8", "HH:mm:ss"); var strTime = String(val).trim(); var match = strTime.match(/(\d{2}:\d{2}:\d{2})/); if (match) return match[1]; match = strTime.match(/(\d{2}:\d{2})/); return match ? match[1] + ":00" : strTime; } function _jsonpResponse(callback, data) { if (callback) { return ContentService.createTextOutput(callback + '(' + JSON.stringify(data) + ')').setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON); } function doOptions(e) { return ContentService.createTextOutput("").setMimeType(ContentService.MimeType.TEXT); }
